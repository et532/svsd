```swift
import Foundation

final class ForceGraph {
    struct Node { let id: String; var x: Double; var y: Double; var vx = 0.0; var vy = 0.0
                  var group = ""; var pinned = false; var degree = 0 }
    struct Link { let a: Int; let b: Int; let strength: Double; let rest: Double }   // indices into nodes
    var nodes: [Node]
    let links: [Link]
    let index: [String: Int]          // id -> node index
    var alpha: Double = 1
    /// prev: earlier positions to keep (web `prev`); seed drives the RNG used where the web calls Math.random.
    init(ids: [String], pairs: [(String, String)], prev: [String: (x: Double, y: Double)] = [:], seed: UInt64 = 1)
    func setGroups(_ groupOf: (String) -> String, last: String?)   // "" = ungrouped; resets alpha to 1
    func step()                                                   // one tick, same math as the web incl. the 640000 cutoff and inner-ring clamp
}

enum MindmapMath {
    static let tierR: [Double] = [8, 19.2, 32]
    static let initialsAt = 0.27, fitZoom = 1.6, maxK = 3.0, grabZoom = 0.6
    static let groupSpread = 3.0, groupPad = 170.0, groupCover = 0.85, other = "Other"
    static let floatK = 55.0, floatC = 8.2, floatMax = 14.0, parallax = 0.07, glideTau = 0.42

    /// web linksOf: unique undirected pairs, only ids present, no self links, first-seen order.
    static func links(_ people: [(id: String, knows: [String])]) -> [(String, String)]
    /// web auraDepth: closeness if set, else 3 if last contact <= 30 days, 2 if <= 120, else 1 (lastContacted is "yyyy-MM-dd", local day).
    static func auraDepth(closeness: Int?, lastContacted: String, now: Date = .now) -> Int
    /// web radiiOf: rank by score desc then id asc; top 22% -> 32, next to 67% -> 19.2, rest -> 8.
    static func radii(_ scores: [(id: String, score: Int)]) -> [String: Double]
    static func disc(_ r: Double) -> Double        // 14 + r * 1.25
    static func dot(_ r: Double) -> Double         // (3.2 + r * 0.15) * 1.4
    static func initialsOpacity(k: Double, fitK: Double) -> Double
    /// web bendOf: JS 32-bit hash over UTF-16 units ((h*31 + c) | 0), then ((|h| % 100)/100 - 0.5) * 0.4. Must equal JS bit for bit (use Int32 wrapping arithmetic, abs in Int64).
    static func bend(_ key: String) -> Double
    static func floatOf(_ id: String, r: Double) -> Double
    static func depthOf(_ id: String, r: Double) -> Double
    /// web groups: value per person; values held by only one person (or empty) go to "Other"; names sorted by size desc,
    /// "Other" appended last if anyone is in it. Empty input value counts as its own value like the web (check the web code).
    static func groups(_ values: [(id: String, value: String)]) -> (groupOf: [String: String], names: [String])
    /// web fit(): view transform (x, y offsets from the screen center, k scale) framing all nodes in a width x height screen.
    static func fit(_ nodes: [ForceGraph.Node], width: Double, height: Double) -> (x: Double, y: Double, k: Double)
    static func minK(fitK: Double) -> Double        // fitK / fitZoom * 0.85
    struct Oval: Equatable { var cx, cy, w, h: Double }
    /// web draw() lines 418-450: mean + spread oval per group, stretch for tight groups, minOf, then grow so every member's disc fits in groupCover.
    static func groupOvals(_ nodes: [ForceGraph.Node], radii: [String: Double]) -> [String: Oval]
}
```
